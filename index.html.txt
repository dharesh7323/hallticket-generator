        // Function to handle registration
        function register() {
            const phoneNumber = document.getElementById('registerPhoneNumber').value;
            const password = document.getElementById('registerPassword').value;
            const confirmPassword = document.getElementById('confirmPassword').value;
            let isValid = true;

            // Clear previous error messages
            document.getElementById('phoneError').innerHTML = '';
            document.getElementById('passwordError').innerHTML = '';
            document.getElementById('confirmPasswordError').innerHTML = '';

            // Validate phone number
            if (phoneNumber.length !== 10 || !/^\d+$/.test(phoneNumber)) {
                document.getElementById('phoneError').innerHTML = 'Please enter a valid 10-digit phone number';
                isValid = false;
            }

            // Check if phone number already exists
            if (mockDatabase.users.some(user => user.phoneNumber === phoneNumber)) {
                document.getElementById('phoneError').innerHTML = 'This phone number is already registered';
                isValid = false;
            }

            // Validate password
            if (password.length < 8) {
                document.getElementById('passwordError').innerHTML = 'Password must be at least 8 characters long';
                isValid = false;
            }

            // Check if passwords match
            if (password !== confirmPassword) {
                document.getElementById('confirmPasswordError').innerHTML = 'Passwords do not match';
                isValid = false;
            }

            if (!isValid) {
                return;
            }

            // Add new user to mock database
            mockDatabase.users.push({
                phoneNumber: phoneNumber,
                password: password,
                role: 'student'
            });

            alert('Registration successful! You can now login with your credentials.');
            backToLogin();
        }

        // Function to simulate login
        function login() {
            const phoneNumber = document.getElementById('phoneNumber').value;
            const password = document.getElementById('password').value;

            // Basic validation
            if (phoneNumber.length !== 10 || !/^\d+$/.test(phoneNumber)) {
                alert('Please enter a valid 10-digit phone number');
                return;
            }

            if (password.length < 8) {
                alert('Password must be at least 8 characters long');
                return;
            }

            // Check if user exists in mock database
            const user = mockDatabase.users.find(u => u.phoneNumber === phoneNumber && u.password === password);
            
            if (!user) {
                alert('Invalid credentials. Please check the demo credentials displayed on the login page or register a new account.');
                return;
            }

            // Set current user
            currentUser = user;

            // Check if user is admin or student
            if (user.role === 'admin') {
                // Load admin dashboard
                loadAdminDashboard();
                // Hide login page and show admin dashboard page
                document.getElementById('loginPage').classList.remove('active');
                document.getElementById('adminDashboardPage').classList.add('active');
            } else {
                // Check if user has an existing application
                const existingApplication = mockDatabase.applications.find(app => app.phoneNumber === phoneNumber);
                if (existingApplication) {
                    currentApplication = existingApplication;
                    checkApplicationStatus();
                } else {
                    // Hide login page and show exam application page
                    document.getElementById('loginPage').classList.remove('active');
                    document.getElementById('examApplicationPage').classList.add('active');
                    // Initialize course options
                    updateCourseOptions();
                }
            }
        }

        // Function to load admin dashboard
        function loadAdminDashboard() {
            const applicationsList = document.getElementById('applicationsList');
            applicationsList.innerHTML = '';

            // Filter pending applications
            const pendingApplications = mockDatabase.applications.filter(app => app.status === 'pending');

            if (pendingApplications.length === 0) {
                applicationsList.innerHTML = '<p>No pending applications.</p>';
                return;
            }

            // Display each application
            pendingApplications.forEach(app => {
                const applicationItem = document.createElement('div');
                applicationItem.className = 'application-item';
                applicationItem.innerHTML = `
                    <div class="application-details">
                        <p><strong>Name:</strong> ${app.fullName}</p>
                        <p><strong>USN:</strong> ${app.usn}</p>
                        <p><strong>Email:</strong> ${app.email}</p>
                        <p><strong>Program:</strong> ${app.program}</p>
                        <p><strong>Branch:</strong> ${app.branch}</p>
                        <p><strong>Semester:</strong> ${app.semester}</p>
                        <p><strong>Courses:</strong> ${app.courses.join(', ')}</p>
                    </div>
                    <div class="admin-actions">
                        <button class="approve-btn" onclick="approveApplication('${app.id}')">Approve</button>
                        <button class="reject-btn" onclick="rejectApplication('${app.id}')">Reject</button>
                    </div>
                `;
                applicationsList.appendChild(applicationItem);
            });
        }

        // Function to approve application
        function approveApplication(applicationId) {
            const applicationIndex = mockDatabase.applications.findIndex(app => app.id === applicationId);
            if (applicationIndex !== -1) {
                mockDatabase.applications[applicationIndex].status = 'approved';
                mockDatabase.applications[applicationIndex].approvalDate = new Date().toISOString();
                alert('Application approved successfully!');
                loadAdminDashboard();
            }
        }

        // Function to reject application
        function rejectApplication(applicationId) {
            const reason = prompt('Please provide a reason for rejection:');
            if (reason !== null) {
                const applicationIndex = mockDatabase.applications.findIndex(app => app.id === applicationId);
                if (applicationIndex !== -1) {
                    mockDatabase.applications[applicationIndex].status = 'rejected';
                    mockDatabase.applications[applicationIndex].rejectionReason = reason;
                    alert('Application rejected successfully!');
                    loadAdminDashboard();
                }
            }
        }

        // Function to submit exam application
        function submitExamApplication(event) {
            event.preventDefault();

            // Collect form data
            const formData = new FormData(event.target);
            const courses = formData.getAll('courses');
            const otherCourses = formData.get('otherCourses');
            
            if (otherCourses && courses.includes('other')) {
                courses.pop(); // Remove 'other'
                courses.push(otherCourses);

            const application = {
                id: generateId(),
                phoneNumber: currentUser.phoneNumber,
                fullName: formData.get('fullName'),
                usn: formData.get('usn'),
                email: formData.get('email'),
                program: formData.get('program'),
                branch: formData.get('branch'),
                semester: formData.get('semester'),
                courses: courses,
                college: formData.get('college'),
                status: 'pending',
                submissionDate: new Date().toISOString()
            };

            // Add application to mock database

            return false;

                    <p><strong>Status:</strong> <span class="status-${currentApplication.status}">${capitalizeFirstLetter(currentApplication.status)}</span></p>
            `;

            if (currentApplication.status === 'approved') {
                statusHTML += `<p><strong>Approval Date:</strong> ${formatDate(currentApplication.approvalDate)}</p>`;

            } else {
                statusHTML += `<p>Your application is currently under review. Please check back later.</p>`;
            }

            statusHTML += `</div>`;
            statusContainer.innerHTML = statusHTML;
        }

        // Helper function to generate a random ID
        function generateId() {

        // Helper function to format date
        function formatDate(dateString) {
            const date = new Date(dateString);
            return date.toLocaleDateString() + ' ' + date.toLocaleTimeString();
        }

        // Helper function to capitalize first letter
        function capitalizeFirstLetter(string) {
            return string.charAt(0).toUpperCase() + string.slice(1);
        }
